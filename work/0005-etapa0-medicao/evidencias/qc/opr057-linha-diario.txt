#!/usr/bin/env bash
# opr057-ensaio.sh <raiz do Tribuna> <pasta nova, fora do repositório> <medir.sh a testar> <rótulo>
# OPR-057: confere a linha do diário "nenhum relatório trocado" onde a troca é recusada. Roda r3-ensaio.sh (T1, T2) e
# r4-ensaio.sh (S0 a S5) sem editá-los e exige a linha uma vez em cada caso. Sai com 1 se algum caso não a tiver.
set -uo pipefail
TRIB="$1"; P="$2"; MEDIR="$3"; ROT="$4"; AQUI="$(cd "$(dirname "$0")" && pwd)"
[ -e "$P" ] && { echo "PARAR: $P já existe; use uma pasta nova" >&2; exit 2; }
mkdir -p "$P"
bash "$AQUI/r3-ensaio.sh" "$TRIB" "$P/r3" "$MEDIR" "$ROT" > "$P/r3.txt" 2>&1
bash "$AQUI/r4-ensaio.sh" "$TRIB" "$P/r4" "$MEDIR" "$ROT" > "$P/r4.txt" 2>&1
echo "=== OPR-057 $ROT: $(shasum -a 256 "$MEDIR" | cut -c1-12) ==="
falhas=0; L='nenhum relatório trocado'
for c in T1 T2; do   # trecho de cada caso: da linha "--- Tn" até a próxima "--- "
  n=$(awk -v c="--- $c:" 'index($0,"--- ")==1{em=(index($0,c)==1);next} em' "$P/r3.txt" | grep -c "$L")
  if [ "$n" = 1 ]; then echo "$c: linha '$L' presente (1): OK"; else echo "$c: linha '$L' x$n (esperado 1): FALHOU"; falhas=$((falhas+1)); fi
done
for c in S0 S1 S2 S3 S4 S5; do
  n=$(grep -c "^$c: medir.sh rc=1; linhas 'nenhum relatório trocado': 1;" "$P/r4.txt")
  if [ "$n" = 1 ]; then echo "$c: linha '$L' presente (1): OK"; else echo "$c: linha '$L' ausente ou repetida: FALHOU"; falhas=$((falhas+1)); fi
done
echo "casos verificados: 8 (T1, T2, S0 a S5); falhas: $falhas"
[ "$falhas" = 0 ] && echo "RESULTADO: VERDE" || echo "RESULTADO: VERMELHO"
[ "$falhas" = 0 ]
